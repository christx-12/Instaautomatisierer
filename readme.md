class VideoGenerator:
    def __init__(self, track_path, video_paths, tacts=4):
        self.track = load_audio(track_path)
        self.videos = [load_video(v) for v in video_paths]
        self.tacts = tacts
        self.bpm = detect_bpm(self.track)
    
    def get_random_startpoint(self):
        """
        Wählt random Startpunkt im Track
        ABER: Nicht die ersten 5 Sekunden (zu früh)
        ABER: Nicht die letzten 10 Sekunden (zu spät)
        """
        min_start = 5000  # 5 Sekunden
        max_start = self.track.duration - 10000  # 10 Sec vor Ende
        
        return random.randint(min_start, max_start)
    
    def generate_initial_video(self):
        """
        Generiert initialen Video mit allen Videos
        """
        start_position = self.get_random_startpoint()
        cut_length_ms = self.tacts_to_milliseconds(
            self.bpm, 
            self.tacts
        )
        
        video_sequence = []
        current_track_position = start_position
        
        for i, video in enumerate(self.videos):
            video_sequence.append({
                'video_file': video,
                'track_start_ms': current_track_position,
                'length_tacts': self.tacts,
                'length_ms': cut_length_ms,
                'order': i
            })
            current_track_position += cut_length_ms
        
        return video_sequence

class EditMode:
    def __init__(self, video_sequence, track):
        self.videos = video_sequence
        self.track = track
    
    def update_video_length(self, video_index, new_tacts):
        """
        Ändert die Länge eines einzelnen Videos in Takten
        """
        old_length = self.videos[video_index]['length_ms']
        new_length = self.tacts_to_milliseconds(128, new_tacts)
        
        self.videos[video_index]['length_tacts'] = new_tacts
        self.videos[video_index]['length_ms'] = new_length
        
        # Recalculate all following videos' start positions
        self.recalculate_positions(video_index)
    
    def update_track_start(self, video_index, new_track_position_ms):
        """
        Verschiebt den Startpunkt eines Videos im Track
        """
        self.videos[video_index]['track_start_ms'] = new_track_position_ms
        # Kann aber auch nicht die nachfolgenden ändern!
        # Jedes Video kann unabhängig started werden
    
    def recalculate_positions(self, from_index):
        """
        Recalculiert automatisch die Positionen nach einer Änderung
        """
        if from_index + 1 < len(self.videos):
            prev_end = (self.videos[from_index]['track_start_ms'] + 
                       self.videos[from_index]['length_ms'])
            self.videos[from_index + 1]['track_start_ms'] = prev_end
    
    def preview_single_video(self, video_index):
        """
        Zeigt Vorschau eines einzelnen Videos mit der Audio
        """
        v = self.videos[video_index]
        audio_segment = self.track[
            v['track_start_ms']: 
            v['track_start_ms'] + v['length_ms']
        ]
        video_segment = load_and_trim_video(
            v['video_file'],
            start=0,
            duration=v['length_ms']
        )
        return sync_audio_video(video_segment, audio_segment)
    
    def preview_full_video(self):
        """
        Zeigt Vorschau des kompletten Videos
        Setzt alle Videos mit ihren Audios zusammen
        """
        final_video = []
        
        for v in self.videos:
            audio_seg = self.track[
                v['track_start_ms']: 
                v['track_start_ms'] + v['length_ms']
            ]
            video_seg = load_and_trim_video(
                v['video_file'],
                duration=v['length_ms']
            )
            final_video.append(sync_audio_video(video_seg, audio_seg))
        
        return concatenate_videos(final_video)

13.03.2026 Konzept
ideo-Editor Architektur: Hybrid-Sync
Konzept:
Das Live-Rendering erfolgt vollständig clientseitig via JavaScript. Die Performance wird durch eine Dual-Video-Strategie und intelligentes Caching sichergestellt.

1. Dual-Video Handling
Jedes Projekt nutzt zwei Versionen der Videodateien:

Raw-Version: Das Original auf dem Server (Single Source of Truth).

Live-Play-Version: Eine optimierte Kopie, die direkt beim Client liegt.

Der Vorteil: Der Client holt sich die Videos einmalig. Danach greift das Browser-Caching: Ein GET-Request auf /uploads/ wird vom Browser instantan aus dem lokalen Cache beantwortet, statt die Datei neu zu laden.

2. Client-Side Rendering Engine

Engine: JavaScript steuert das Rendering live im Browser.

Layer-Stack: Alle Clips der Timeline werden als simultane <video>-Layer übereinander geladen.

Frame-Sync: Ein Master-Loop (requestAnimationFrame) toggelt die Sichtbarkeit (opacity) der Layer basierend auf dem globalen Timehead. Dies eliminiert Umschaltpausen ("Gaps").

3. Timeline & Synchronisation

Zustand: Eine JSON-Struktur beschreibt die Timeline (Welches Video? Welche Sekunde? Welcher Pfad?).

Synchronität: Diese Metadaten werden permanent zwischen Client und Server abgeglichen. So bleibt die Bearbeitung über Sessions hinweg konsistent, während das schwere Videomaterial lokal gecached bleibt.
